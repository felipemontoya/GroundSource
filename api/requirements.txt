# Runtime dependencies for the GroundSource backend.
#
# Every entry needs a reason (see ../AGENTS.md). Pinned to a compatible
# release so that a rebuild is reproducible without a lockfile yet.

# The web framework itself. 5.2 is the current LTS.
Django~=5.2.0

# PostgreSQL driver. psycopg 3 is what Django 5 targets; the binary wheel
# avoids needing libpq and a C toolchain in the runtime image.
psycopg[binary]~=3.3

# ASGI server. The stack runs on ASGI from the start because token
# streaming (SSE) is a Phase 1 requirement and switching servers later
# would invalidate the local proxy path this environment exists to exercise.
# The [standard] extra brings uvloop/httptools (the usual production choice)
# and watchfiles, which is what makes --reload work in the dev container.
uvicorn[standard]~=0.52.0

# pgvector's Django integration: a VectorField and index classes, so the
# vector column is described in migrations like every other column instead
# of in hand-written SQL scattered through the app.
pgvector~=0.5.0

# PDF extraction. MIT, built on pdfminer.six, and it exposes every
# character's page and bounding box — which is the provenance the offset
# invariant needs. Docling was measured instead of assumed: it installs
# 6.2 GB (3.2 GB of it CUDA) for a layout model this born-digital document
# does not need. See grounding/extraction.py for the full reasoning.
pdfplumber~=0.11.10

# Embeddings and chat completion. One provider, behind grounding/llm.py, so
# that the provider choice stays a single file rather than a commitment.
openai~=3.13.0
